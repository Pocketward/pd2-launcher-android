# Debian 13 (trixie) arm64 packages of the App userland, at the snapshot in config.sh (DEBIAN_SNAPSHOT).
# fetch-debian.sh --resolve turns this list into packages.lock (exact dependency closure, SHA-256 per .deb).
#
# Line format:  <set> <package>=<version>   or   exclude <package>
#   runtime  goes into the App userland (and the build sysroot)
#   build    goes only into the cross-build sysroot (headers and .pc files for build-xvfb.sh, build-panvk.sh)
#   exclude  dropped from the closure: needed only by maintainer scripts, which are never run

runtime libpixman-1-0=0.44.0-3
