# Android-blocked arm64 host syscalls that FEX answers with -EPERM instead of -ENOSYS.
# Input of gen-blocked-host-syscalls.py, together with blocked-host-syscalls.txt (the blocked list).
#
# These are the privileged calls: an unprivileged process on Linux gets EPERM for them, so -EPERM is
# what guest programs already expect from a normal user.
# The set*id calls are listed so that a raw passthrough of them fails safe; FEX patch 0001 answers the
# guest set*id syscalls with its own handlers (0 for a no-op, else -EPERM).
#
# Every other blocked syscall is answered with -ENOSYS.

# mount family
mount
umount2
open_tree
move_mount
fsopen
fsconfig
fsmount
fspick
mount_setattr

# system administration
chroot
acct
swapon
swapoff
syslog
reboot
settimeofday
clock_settime
clock_adjtime
adjtimex
sethostname
setdomainname
vhangup
quotactl_fd

# kernel modules and kexec
init_module
finit_module
delete_module
kexec_load
kexec_file_load

# set*id (blocked by SECCOMP_BLOCKLIST_APP.TXT; setresuid stays allowed)
setuid
setgid

setregid

setfsgid
